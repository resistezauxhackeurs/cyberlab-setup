import base64
import hmac
import hashlib

jeton="eyJ0eXAiOiJKV1QiLCJhbGciOiJIUzI1NiJ9Cg.eyJsb2dpbiI6ImFkbWluIn0K"

f = open("public.pem")
key = f.read()

sig = base64.urlsafe_b64encode(hmac.new(key,jeton,hashlib.sha256).digest()).decode('UTF-8').rstrip("=")

print (jeton+"."+sig)
